\begin{problem}[Warmup]{(a)}
    Is every abelian group solvable? If not, give an example of a non-solvable abelian group.
\end{problem}
\unsolved

\begin{problem}[Warmup]{(b)}
    Is every solvable group abelian? If not, give an example of a non-abelian solvable group.
\end{problem}
\unsolved

\begin{problem}[Warmup]{(c)}
    Show that the group of quaternions $Q_8$ is solvable. That is, give an appropriate subnormal series for this group.
\end{problem}
\unsolved

\begin{problem}{1}
    Suppose that $G$ and $H$ are solvable groups. Prove that $G \times H$ is solvable.
\end{problem}
\unsolved

\begin{problem}{3}
    Suppose the $G$ is a finite abelian group. Prove that $G$ has a composition series;
    that is, show that $G$ has a collection of subgroups $G_i$ so that
    $G_n = \{1\} \subseteq G_{n - 1} \subseteq \cdots \subseteq G_1 \subseteq G_0 = G$, and
    $G_{i - 1}/G_i$ is a simple group.
\end{problem}
\unsolved

\begin{problem}{4}
    Suppose that $G$ is a finite solvable group. Prove that $G$ has a composition series.
\end{problem}
\unsolved

\begin{problem}{5}
    Prove that $\mathbb{Z}$ does not have a composition series.
\end{problem}
\unsolved

\begin{problem}{6}
    Let $G$ be a group and $H$ a proper normal subgroup. Prove that $G/H$ is a simple group
    if and only if $H$ is a maximal normal subgroup.
\end{problem}
\unsolved

\begin{problem}{7}
    Generalize Exercise 36.4. That is, prove that every finite group has a composition
    series.
\end{problem}
\unsolved
